   181	function base64UrlToBytes(value) {
   182	  const normalized = String(value || "").replace(/-/g, "+").replace(/_/g, "/");
   183	  const padded = normalized + "=".repeat((4 - (normalized.length % 4)) % 4);
   184	  const binary = atob(padded);
   185	  const bytes = new Uint8Array(binary.length);
   186	  for (let index = 0; index < binary.length; index += 1) bytes[index] = binary.charCodeAt(index);
   187	  return bytes;
   188	}

   190	async function hashPassword(password) {
   191	  const salt = randomBytes(16);
   192	  const material = await crypto.subtle.importKey("raw", new TextEncoder().encode(password), "PBKDF2", false, ["deriveBits"]);
   193	  const iterations = 100000;
   194	  const bits = await crypto.subtle.deriveBits({ name: "PBKDF2", salt, iterations, hash: "SHA-256" }, material, 256);
   195	  return `pbkdf2$${iterations}$${bytesToBase64Url(salt)}$${bytesToBase64Url(new Uint8Array(bits))}`;
   196	}

   198	async function verifyPassword(password, stored) {
   199	  const [scheme, iterationsText, saltText, digestText] = String(stored || "").split("$");
   200	  if (scheme !== "pbkdf2" || !iterationsText || !saltText || !digestText) return false;
   201	  const material = await crypto.subtle.importKey("raw", new TextEncoder().encode(password), "PBKDF2", false, ["deriveBits"]);
   202	  const bits = await crypto.subtle.deriveBits({ name: "PBKDF2", salt: base64UrlToBytes(saltText), iterations: Math.min(Number(iterationsText) || 100000, 100000), hash: "SHA-256" }, material, 256);
   203	  return bytesToBase64Url(new Uint8Array(bits)) === digestText;
   204	}

   207	  if (!env.NEXUS_CREDENTIALS_KEY) throw new Error("Credential encryption is not configured");
   208	  const keyBytes = base64UrlToBytes(env.NEXUS_CREDENTIALS_KEY);
   209	  if (keyBytes.byteLength !== 32) throw new Error("Credential encryption key must be 32 bytes");
   210	  const key = await crypto.subtle.importKey("raw", keyBytes, { name: "AES-GCM" }, false, ["encrypt"]);
   211	  const iv = randomBytes(12);

   608	function pemToArrayBuffer(pem) {
   609	  const clean = pem.replace(/-----BEGIN PRIVATE KEY-----/g, "").replace(/-----END PRIVATE KEY-----/g, "").replace(/\s/g, "");
   610	  const binary = atob(clean);
   611	  const bytes = new Uint8Array(binary.length);
   612	  for (let i = 0; i < binary.length; i++) bytes[i] = binary.charCodeAt(i);
   613	  return bytes.buffer;
   614	}

   616	async function googleAccessToken(serviceAccountJson) {
   617	  const account = typeof serviceAccountJson === "string" ? JSON.parse(serviceAccountJson) : serviceAccountJson;
   618	  if (!account?.client_email || !account?.private_key) throw new Error("Google service account secret is missing client_email or private_key");
   619	
   620	  const now = Math.floor(Date.now() / 1000);
   621	  const header = textBase64Url(JSON.stringify({ alg: "RS256", typ: "JWT" }));
   622	  const claim = textBase64Url(JSON.stringify({
   623	    iss: account.client_email,
   624	    scope: "https://www.googleapis.com/auth/spreadsheets",
   625	    aud: "https://oauth2.googleapis.com/token",
   626	    iat: now,
   627	    exp: now + 3600,
   628	  }));
   629	  const unsignedToken = `${header}.${claim}`;
   630	  const key = await crypto.subtle.importKey(
   631	    "pkcs8",
   632	    pemToArrayBuffer(account.private_key),
   633	    { name: "RSASSA-PKCS1-v1_5", hash: "SHA-256" },
   634	    false,
   635	    ["sign"]
   636	  );
   637	  const signature = await crypto.subtle.sign("RSASSA-PKCS1-v1_5", key, new TextEncoder().encode(unsignedToken));
   638	  const assertion = `${unsignedToken}.${base64Url(new Uint8Array(signature))}`;
   639	  const tokenResponse = await fetch("https://oauth2.googleapis.com/token", {
   640	    method: "POST",
   641	    headers: { "content-type": "application/x-www-form-urlencoded" },
   642	    body: new URLSearchParams({ grant_type: "urn:ietf:params:oauth:grant-type:jwt-bearer", assertion }),
   643	  });
   644	  const tokenData = await tokenResponse.json();
   645	  if (!tokenResponse.ok || !tokenData.access_token) throw new Error(`Google OAuth failed: ${tokenData.error_description || tokenData.error || "unknown error"}`);
   646	  return tokenData.access_token;
   647	}

   669	async function ensureSheetTab(sheetId, sheetTab, headers, authHeaders, cleanupLegacy = false) {
   670	  const headerRange = `${sheetTab}!A1:${String.fromCharCode(64 + headers.length)}1`;
   671	  const headerUrl = new URL(`https://sheets.googleapis.com/v4/spreadsheets/${encodeURIComponent(sheetId)}/values/${encodeURIComponent(headerRange)}`);
   672	  headerUrl.searchParams.set("majorDimension", "ROWS");
   673	  let headerResponse = await fetch(headerUrl, { headers: authHeaders });
   674	  let headerData = await headerResponse.json();
   675	
   676	  if (!headerResponse.ok && headerData.error?.message?.includes("Unable to parse range")) {
   677	    const createUrl = `https://sheets.googleapis.com/v4/spreadsheets/${encodeURIComponent(sheetId)}:batchUpdate`;
   678	    const createResponse = await fetch(createUrl, {
   679	      method: "POST",
   680	      headers: authHeaders,
   681	      body: JSON.stringify({ requests: [{ addSheet: { properties: { title: sheetTab } } }] }),
   682	    });
   683	    const createData = await createResponse.json();
   684	    if (!createResponse.ok) throw new Error(`Google Sheets tab creation failed: ${createData.error?.message || "unknown error"}`);
   685	    headerData = { values: [] };
   686	  }
   687	  if (!headerResponse.ok && !headerData.values) throw new Error(`Google Sheets header check failed: ${headerData.error?.message || "unknown error"}`);
   688	
   689	  const existingHeader = headerData.values?.[0] || [];
   690	  const headerMatches = headers.every((header, index) => existingHeader[index] === header);
   691	  if (cleanupLegacy && LEGACY_COMBINED_HEADERS.every((header, index) => existingHeader[index] === header)) {
   692	    const clearUrl = new URL(`https://sheets.googleapis.com/v4/spreadsheets/${encodeURIComponent(sheetId)}/values/${encodeURIComponent(`${sheetTab}!O:Z`)}:clear`);
   693	    const clearResponse = await fetch(clearUrl, { method: "POST", headers: authHeaders, body: "{}" });
   694	    const clearData = await clearResponse.json();
   695	    if (!clearResponse.ok) throw new Error(`Google Sheets legacy draft cleanup failed: ${clearData.error?.message || "unknown error"}`);
   696	  }
   697	  if (!headerMatches) {
   698	    const updateUrl = new URL(`https://sheets.googleapis.com/v4/spreadsheets/${encodeURIComponent(sheetId)}/values/${encodeURIComponent(headerRange)}`);
   699	    updateUrl.searchParams.set("valueInputOption", "USER_ENTERED");
   700	    const updateResponse = await fetch(updateUrl, {
   701	      method: "PUT",
   702	      headers: authHeaders,
   703	      body: JSON.stringify({ majorDimension: "ROWS", values: [headers] }),
   704	    });
   705	    const updateData = await updateResponse.json();
   706	    if (!updateResponse.ok) throw new Error(`Google Sheets header update failed: ${updateData.error?.message || "unknown error"}`);
   707	  }
   708	  return { header_written: !headerMatches };
   709	}

   806	function extractEmail(text, website = "") {
   807	  if (text) {
   808	    const matches = text.match(/[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,24}/gi) || [];
   809	    const ignored = new Set(["example@example.com", "email@example.com", "name@example.com"]);
   810	    const blockedTlds = new Set(["png", "jpg", "jpeg", "gif", "svg", "webp", "bmp", "ico", "pdf"]);
   811	    const candidate = matches.map((item) => item.toLowerCase().replace(/[),.;:]+$/g, ""))
   812	      .find((item) => {
   813	        const [local, domain] = item.split("@");
   814	        const tld = domain?.split(".").pop() || "";
   815	        const firstLabel = domain?.split(".")[0] || "";
   816	        return Boolean(local && domain && domain.includes(".") && /^[a-z]{2,24}$/.test(tld))
   817	          && !blockedTlds.has(tld)
   818	          && !ignored.has(item)
   819	          && !item.includes("sentry")
   820	          && !item.includes("wixpress")
   821	          && !/^(?:[0-9]+x|image|img)\./i.test(`${firstLabel}.`)
   822	          && !/(?:^|[_-])\d{2,}x(?:[_-]|$)/i.test(local);
   823	      });
   824	    if (candidate) return candidate;
   825	  }
   826	  if (website) {
   827	    try {
   828	      const parsed = new URL(website);
   829	      const domain = parsed.hostname.replace(/^www\./, "");
   830	      if (domain && !domain.includes("localhost") && !domain.includes("google") && !domain.includes("facebook")) {
   831	        return `info@${domain}`;
   832	      }
   833	    } catch {}
   834	  }
   835	  return null;
   836	}

   838	function extractSocialLinks(text, html = "") {
   839	  const combined = `${text} ${html}`;
   840	  const getUrl = (regex) => {
   841	    const match = combined.match(regex);
   842	    if (!match) return null;
   843	    let url = match[0].replace(/[),.;:]+$/g, "");
   844	    if (!url.startsWith("http")) url = `https://${url}`;
   845	    try {
   846	      const parsed = new URL(url);
   847	      return parsed.toString();
   848	    } catch {
   849	      return null;
   850	    }
   851	  };
   852	
   853	  const facebook = getUrl(/https?:\/\/(?:www\.)?(?:facebook|fb)\.com\/[A-Za-z0-9._%+-]+/i);
   854	  const instagram = getUrl(/https?:\/\/(?:www\.)?instagram\.com\/[A-Za-z0-9._%+-]+/i);
   855	  const twitter = getUrl(/https?:\/\/(?:www\.)?(?:twitter|x)\.com\/[A-Za-z0-9._%+-]+/i);
   856	  const linkedin = getUrl(/https?:\/\/(?:www\.)?linkedin\.com\/(?:company|in)\/[A-Za-z0-9._%+-]+/i);
   857	
   858	  return { facebook, instagram, twitter, linkedin };
   859	}
